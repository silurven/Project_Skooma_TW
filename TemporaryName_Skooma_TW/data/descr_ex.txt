; Sound — rolloff controls 3D distance attenuation
; Lower values = sounds carry further
; OG: 0.4 (designed for a zoomed-in camera)
rolloff_factor 0.25

; Strategy map sound rolloff (same scale as above)
stratmap_rolloff_factor 0.4

; Doppler pitch-shift from camera movement (0 = off, 1 = realistic)
doppler_factor 0.3

; Maximum number of factions (default in M2: 31, RTW: 21)
; Increase to support more factions in mods
max_factions 100

; Battle camera height limits (above terrain, in world units)
; Min: how close you can zoom in (default 0.1, floor 0.1)
; Max: how far you can zoom out (default 100.0)
camera_min_height 0.1
camera_max_height 100.0

; Minimum seconds the main loading screen must stay visible before closing.

;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
; Battle visuals
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

; Maximum blood pools on the battlefield at once (default 192, max 2048).
; Oldest pool fades out when the limit is reached. 0 turns blood pools off.
; Purely visual so any value is safe for multiplayer.
blood_pool_limit 192

; Maximum corpses on the battlefield at once (default 1600).
; Above the limit, the oldest corpse on the field is removed.
; This is purely visual in Medieval 2 - corpses don't block anything,
; so any value is safe for multiplayer.
corpse_limit 1600

; Minimum right click drag distance before formation indicator is rendered.
; Default lowered from 100 > 25 to be more responsive.
min_rightclick_drag 25.0

; UI unit flash button duration in seconds, float range(1.0-600.0).
unit_flash_duration 5.0

; Controls how often the UI unit flash occurs in seconds, range(1.0-600.0).
; Setting lower than unit_flash_duration will make it flash slower.
; Cannot be set higher than unit_flash_duration or they will be forced to match.
unit_flash_frequency 5.0

; Colour alpha of the UI unit flash colour, range(0-127).
unit_flash_intensity 127

; show_waypoints hotkey(default space) also highlights all units when held.
; Uses same unit_flash_intensity setting to set alpha
space_unit_highlight true
